      -- never: dependency 'unlimited-reforge'
local unlimitedRes = nil

local function FindUnlimited()
    for i = 0, GetNumResources() - 1 do
        local res = GetResourceByFindIndex(i)
        if res and GetResourceState(res) == 'started'
            and GetResourceMetadata(res, 'author', 0) == 'Unlimited Reforged-Team' then
            return res
        end
    end
    return nil
end

CreateThread(function() Wait(1500) unlimitedRes = FindUnlimited() end)

AddEventHandler('onResourceStart', function(res)
    if res ~= GetCurrentResourceName()
        and GetResourceMetadata(res, 'author', 0) == 'Unlimited Reforged-Team' then
        SetTimeout(1500, function() unlimitedRes = FindUnlimited() end)
    end
end)

AddEventHandler('onResourceStop', function(res)
    if res == unlimitedRes then unlimitedRes = nil end
end)

-- every call to Unlimited goes through a guard like this
local function CallUnlimited(fn, ...)
    if not unlimitedRes or GetResourceState(unlimitedRes) ~= 'started' then return end
    local args = { ... }
    pcall(function() exports[unlimitedRes][fn](exports[unlimitedRes], table.unpack(args)) end)
end

      -- server/unlimited.lua (shortened). `allowed` = the caller is the linked Unlimited
local function allowed(caller) return caller ~= nil and caller == unl end

local function safe(fn, ...)            -- never let an error out of an export
    local out = table.pack(pcall(fn, ...))
    if not out[1] then return false, 'error' end
    return table.unpack(out, 2, out.n)
end

exports('UnlRemoveItem', function(kind, id, name, count, slot)
    if not allowed(GetInvokingResource()) then return false, 'not_unlimited' end
    return safe(function()
        local inv = resolve(kind, id)               -- your own lookup
        if not inv then return false, 'bad_inventory' end
        local ok, err = Inv.removeFrom(inv, name, count, slot)
        if not ok then return false, err end
        Inv.save(inv)                               -- your cache, rules and screens stay correct
        return true
    end)
end)

-- tell Unlimited after every save, and after the item registry changed (only while linked)
if unl then TriggerEvent('unlimited:inventory:changed', GetCurrentResourceName(), kind, tostring(id)) end
if unl then TriggerEvent('unlimited:inventory:items', GetCurrentResourceName()) end
